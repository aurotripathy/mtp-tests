Parse ISO-8601 timestamps without using the datetime module.

This module provides a function to parse ISO-8601 formatted timestamps
and return a tuple of (year, month, day, hour, minute, second, microsecond, tz_offset_minutes).
"""

import re
from typing import Tuple, Optional

    - YYYY-MM-DDTHH:MM:SS.ssssss-HH:MM
    - YYYY-MM-DDTHH:MM:SS.ssssssZ

    - YYYY-MM-DDTHH:MM:SS-HH:MM
    - YYYY-MM-DDTHH:MM:SSZ
    
    Args:
        timestamp: An ISO-8601 formatted timestamp string.
        
    Returns:
        A tuple of (year, month, day, hour, minute, second, microsecond, tz_offset_minutes)
        where tz_offset_minutes is None if no timezone offset is present,
        or the offset in minutes (positive for east of UTC, negative for west).
